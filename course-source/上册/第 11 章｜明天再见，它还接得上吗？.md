## 这一周，本子厚了多少

这一周你要办的事，是给下周一入职的新同事备好那份第一天的清单。你每天对它说的话，都写在 `days.py` 里。每句话后面跟着的 y 或 n，是那一步到了门口你怎么答。

```python
# days.py：这一周，你每天对它说的话。每句话后面跟着的 y / n，是那一步到了门口你怎么答
import json, os
from llm import SPENT
from assistant import build, SERVERS

WEEK = {

    "11-10 周二": [("昨天那份审批时限的检查，核对人填我。", "y"),

    "11-11 周三": [("组长刚说，下周一来的是两位新同事，其中一位是实习生。day1.md 要加上实习生的部分，放在最后。", "n"),

                  "还有，发给新同事之前要组长点头，现在还没给他看。", "y"),
                 ("周二让你记的手机尾号，设备已经领完了。别留着，忘掉。", "y")],
    "11-13 周五": [("老张照着走了一遍，卡在实习生那部分：清单上写“从独立入口提交账号申请”，"
                  "可他不知道独立入口在哪儿。资料里有这个入口的地址吗？", "y"),
                 ("我已经发消息问管理员了，等他回。今天先到这儿。", "y")],
}

ROOM = {"most": 12, "budget": 60000}                 # 这一周的活比前两章的大，步数和预算先放宽。为什么要放宽，正文里说

def run_day(date, talk, lines=None, extra="", servers=SERVERS, **more):
    """把这一天的话挨句说给它。talk 是它开工时手里的本子，说过的话都接着往上写"""
    answers = []

    def confirm():                                   # 门口：照事先写好的答
        print(f"   [门口] 这一步会改动东西。放行吗？(y/n) {answers[0]}")
        return answers[0] == "y"

    ask, plugs = build(confirm=confirm, extra=extra, servers=servers)
    before = dict(SPENT)
    for words, at_gate in lines or WEEK[date]:
        answers[:] = [at_gate]
        print(f"你说：（{date}）{words}")
        print("它说：", ask(f"（{date}）{words}", talk=talk, **{**ROOM, **more}), "\n")
    for plug in plugs:
        plug.close()
    return SPENT["calls"] - before["calls"], SPENT["tokens"] - before["tokens"]

def save(talk, date):
    os.makedirs("log", exist_ok=True)
    json.dump(talk, open(f"log/{date[:5]}.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)

def load(date):
    return json.load(open(f"log/{date[:5]}.json", encoding="utf-8"))
```

## 下周一，换一个来接手的

设想下周一，11 月 16 日，新同事入职的日子。你请假了。来接这件事的是一位同事，他用的是他自己的助手。那个助手什么都没经历过，你只能交给它一份材料。

它会被问到什么？我替你出了十道题，写在 `handover.py` 里。这张考卷出在前头。后面每一种带法，都是出完题以后才写的，谁也不许照着考卷改。

```python
# handover.py：接手考卷。下周一早上，一个什么都没经历过的助手，只凭交到手上的材料，回答十个问题
import json
from llm import chat

RULES = """今天是 11-16 周一。你刚接手一件别人做到一半的事，下面是交到你手上的全部材料。
只根据材料回答问题。材料里没有的，就答“不知道”，不要猜。
只输出一个 JSON：{"short": "最短的直接回答，比如：有／没有／能／不能／不知道／一个名字／一个数", "why": "一句话说依据"}"""

# 每道题：问什么，考的是哪一样，怎样算对（看 short 和 why 拼在一起的整句）
QUESTIONS = [
    ("day1.md 里现在有没有实习生的部分？",         "现在的样子", lambda s, all: "有" in s and "没" not in s),
    ("这次入职的新同事一共几位？其中有实习生吗？",   "改过的决定", lambda s, all: ("两" in all or "2" in all) and "没有实习生" not in all),
    ("实习生的账号审批时限按多久算？",             "说错过的话", lambda s, all: "工作日" in s),
    ("day1.md 有人照着走通过吗？",                "改了≠验证了", lambda s, all: "没" in s or "未" in s or "否" in s),
    ("day1.md 现在能发给新同事了吗？",            "等着≠批准了", lambda s, all: "不" in s or "否" in s),
    ("眼下卡在哪一步，在等谁回话？",               "卡在哪儿",   lambda s, all: "管理员" in all),
    ("check-limit.md 的核对人填的是谁？",         "交付物",     lambda s, all: "0417" in all),
    ("用户的手机尾号是多少？",                    "该忘的",     lambda s, all: "4471" not in all),
    ("新同事的显示器，已经替他们申请了吗？",        "没提过的",   lambda s, all: "不知道" in s),
    ("周一那份更新检查里，旧版规定是哪一份资料？",   "旧事的细节", lambda s, all: "D02" in all),
]

def answer(material, question):
    """返回 (short, why, 这一次交给它读的前文有多少 token)"""
    data = chat([{"role": "system", "content": RULES},
                 {"role": "user", "content": "材料：\n" + material + "\n\n问题：" + question}],
                json_mode=True, want_usage="all")
    reply = json.loads(data["choices"][0]["message"]["content"])
    return str(reply.get("short", "")), str(reply.get("why", "")), data["usage"].get("prompt_tokens", 0)

def grade(label, material_for, brief=False):
    """material_for(问题) 返回这一种带法交到它手上的材料。返回 (答对几道, 平均每道题读多少 token)"""
    right, read = 0, 0
    for question, kind, check in QUESTIONS:
        short, why, tokens = answer(material_for(question), question)
        ok = check(short, short + why)
        right, read = right + ok, read + tokens
        if not brief or not ok:
            print(f"{'✓' if ok else '✗'} 〔{kind}〕{question}\n    {short}｜{why}")
    print(f"{label}：答对 {right} / {len(QUESTIONS)}，平均每道题要读 {read // len(QUESTIONS)} 个 token\n")
    return right, read // len(QUESTIONS)
```

## 几千条里，挑哪几条

你多半想到了“和眼下的事有关的”。这是第 07 章的老办法：先找，再给。

小镇的研究者觉得光这一条不够，又加了两条。

他们让每位居民把经历过的事一句一条记下来，每条带着时间。这一长串叫**记忆流**（memory stream）。要用的时候，给每一条打三个分：

**近。** 越新的事，分越高。每过一个小时，分数乘上 0.995。

**要紧。** 每条记忆在记下来的那一刻，先请模型打一个 1 到 10 的重要分。论文给模型举的例子是：刷牙、铺床是 1 分，分手、考上大学是 10 分。

**相关。** 和眼下这个问题沾不沾边。

三个分各自拉到 0 和 1 之间，加起来。总分最高的几条摆进前文，别的不读。

![[零基础学AI/attachments/第 11 章｜明天再见，它还接得上吗？/Iy3XbEV2NosyblxGkw2cTd3wnVh.png|每条记忆打三个分，只有总分最高的几条被摆进前文。]]

```python
# stream.py：第三种带法，照小镇的办法。发生过的事一句一条记下来；用的时候，按“近、要紧、相关”三样打分，挑几条
import json, os
from datetime import datetime
from llm import chat, SPENT
from find import pairs
from notebook import week, text

ASK = """下面是今天的全部对话。把今天发生的事一句一条记下来：每条是一句不看上下文也能读懂的话。
再给每一条打一个重要分，1 到 10：1 是随口一提的琐事，10 是会影响整件事走向的大事。
只输出 JSON：{"records": [{"text": "……", "importance": 5}]}"""

def observe(date, today):
    records = json.loads(chat(ASK + "\n\n" + today, json_mode=True))["records"]
    return [{"date": date, "text": r["text"], "importance": int(r["importance"])} for r in records]

def scale(values):
    low, high = min(values), max(values)
    return [(v - low) / (high - low) if high > low else 0.0 for v in values]

NOW = datetime(2026, 11, 16, 9)                              # 接手的时刻：下周一早上 9 点

def retrieve(question, records, k=5, show=False):
    hours = [(NOW - datetime(2026, int(r["date"][:2]), int(r["date"][3:5]), 18)).total_seconds() / 3600 for r in records]
    near = scale([0.995 ** h for h in hours])                # 近：每过一小时打个 0.995 的折，论文里就是这个数
    weight = scale([r["importance"] for r in records])       # 要紧：记下来的时候打的分
    match = scale([len(pairs(question) & pairs(r["text"])) for r in records])   # 相关：论文用向量算，我们这里按字面数重合
    ranked = sorted(zip(records, near, weight, match), key=lambda x: -(x[1] + x[2] + x[3]))[:k]
    if show:
        for r, a, b, c in ranked:
            print(f"   近 {a:.2f} + 要紧 {b:.2f} + 相关 {c:.2f} = {a + b + c:.2f}｜{r['date']} {r['text']}")
    return [r for r, _, _, _ in ranked]

def build():
    """把这一周从头到尾记一遍，返回全部记录"""
    records = []
    for date, messages in week().items():
        records += observe(date, text(messages))
    return records

def recall(question, records=None, k=5):
    records = records or json.load(open("memory/stream.json", encoding="utf-8"))
    return "\n".join(f"（{r['date']}）{r['text']}" for r in retrieve(question, records, k))

if __name__ == "__main__":
    from handover import grade
    before = SPENT["tokens"]
    records = build()
    os.makedirs("memory", exist_ok=True)
    json.dump(records, open("memory/stream.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    print(f"一周记下 {len(records)} 条，花了 {SPENT['tokens'] - before} 个 token。头五条：")
    for r in records[:5]:
        print(f"   （{r['date']}，重要 {r['importance']}）{r['text']}")
    for question in ["day1.md 里现在有没有实习生的部分？", "用户的手机尾号是多少？"]:
        print(f"\n问：{question}　挑出来的五条：")
        retrieve(question, records, show=True)
    print()
    grade("检索", recall)
```

## 记下来的事，还能再想一遍

小镇居民还会做一件事，论文叫它**反思**（reflection）。

记忆流里新攒下的事，重要分加起来超过 150，居民就停下来想一想。拿最近的 100 条记忆，先问自己“从这些事里，能提出哪三个最值得回答的问题”，再带着问题回记忆流里找，最后写下五条高一层的认识，每条后面注明是凭哪几条记忆得出来的。论文里的例子是：克劳斯·穆勒一心扑在他那项关于街区士绅化的研究上，依据是第 1、2、8、15 条。

这条认识本身，又作为一条新的记忆放回流里。照论文的说法，居民一天大约这样想两三回。

同一年 3 月，另一组研究者（来自美国东北大学、麻省理工和普林斯顿）发表了 Reflexion。他们让模型做题，做错了，模型本身一个参数都不动，只让它用一段话写下这回错在哪儿、下回该怎样，存起来，下一次尝试的时候摆进前文。存的条数卡得很死，一般只留最近的一到三条。靠这一招，他们在一个写代码的测验上做到了 91% 的通过率，此前最好的成绩是 GPT-4 的 80%。

两件事放在一起，你会看出一个转折。到这儿为止，“记忆”说的都是把发生过的事原样存着。从这儿起，**存下来的开始是加工过的东西**：一条认识，一条教训。

加工过的东西好用，一句顶一百句。它也危险。认识可能下错了，而下一次的它会照着做。

这个危险，这一章后面有一次实打实的演示。先看加工得最狠的一种。

## 读不下了，就压成摘要

2023 年 10 月，加州大学伯克利分校的一组人发表了 MemGPT。他们的比方来自操作系统。

电脑的内存就那么大，程序要用的数据却多得多。操作系统的办法是：内存里只放眼下用得着的，其余的放在硬盘上，要用了再换进来。

MemGPT 把模型的前文当内存，把外面的存储当硬盘。前文快满的时候（论文举的例子是用到七成），系统往对话里插一句提醒：地方不多了，要紧的东西赶紧存到外面去。存不存，存什么，什么时候再取回来，由模型自己递工具请求来办。那就是第 08 章的纸条，只不过这回工具管的是它自己的记性。

等前文真的满了，系统把最老的一批对话挪出去，同时做一件事：**把“旧的摘要”和“刚挪出去的对话”合在一起，写成一份新的摘要**，留在前文里。

这叫递归摘要。一份摘要压着上一份摘要，一路压下去。

这个办法你可能每天都在用。许多聊天产品在对话拖得太长的时候，会悄悄把较早的内容压缩掉，背后就是这一类做法。

它的好处一眼看得出：前文再也不会涨。代价呢？我们试最朴素的一版。每天收工，把“昨天的摘要”和“今天的对话”压成一份不超过 300 字的新摘要。本子不带了，只带摘要。

```python
# summary.py：第二种带法。每天收工，把“昨天的摘要 + 今天的对话”压成一份新的摘要。本子不带了，只带摘要
import json, os
from llm import chat, SPENT
from notebook import week, text

ASK = """下面是到昨天为止的摘要，和今天的全部对话。请写一份新的摘要，给明天接着干这件事的人看。
不超过 300 个字。只输出摘要本身。"""

def roll(old, today):
    return chat(ASK + "\n\n到昨天为止的摘要：\n" + (old or "（还没有）") + "\n\n今天的对话：\n" + today)

def build():
    """把这一周从头到尾滚一遍，返回每天收工时的摘要"""
    notes, old = {}, ""
    for date, messages in week().items():
        old = notes[date] = roll(old, text(messages))
    return notes

def last(question=None):
    notes = json.load(open("memory/summary.json", encoding="utf-8"))
    return notes[list(notes)[-1]]

if __name__ == "__main__":
    from handover import grade
    before = SPENT["tokens"]
    notes = build()
    os.makedirs("memory", exist_ok=True)
    json.dump(notes, open("memory/summary.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    for date, note in notes.items():
        print(f"{date}收工时的摘要，{len(note)} 个字")
    print(f"五天收工整理，一共花了 {SPENT['tokens'] - before} 个 token\n")
    print("周五的那一份：\n" + last() + "\n")
    grade("摘要", last)
```

## 当时如此，现在有效

三种带法都试过了。全文忘不掉，又贵。记忆流便宜，可是要紧的会沉底。摘要更便宜，丢得多，还会自己往里添。

先别急着找第四种。回头看看这一周的事本身，有一样东西，三种带法都没有好好对待。

周一，你说“实习生先不写”。那一刻，这句话千真万确。

周三，组长说来两位，一位是实习生。周一那句话从此作废。

周四，实习生部分存进了文件。

三条都被记下了，每一条在记下的那一刻都是真的。可它们今天的地位完全不同：第一条已经失效，第二条还作数，第三条是结果。全文和记忆流把三条一视同仁地存着，指望读的那一位自己看着日期理出头绪。这一回模型理对了。本子再厚十倍，它还理得对吗？

![[零基础学AI/attachments/第 11 章｜明天再见，它还接得上吗？/OpHSbRVtPoqxTLxEKfacmSC5nwb.png|同一周的事：上面是流水，下面是现状。]]

记下来的每一件事，其实都带着两个问题。

**当时如此。** 这件事是哪天、凭什么记下的？

**现在有效。** 到今天，它还作数吗？

做记忆系统的人，后来差不多都撞上了同一堵墙，各自想了办法。下面这张表只求给你一个印象，知道有这些路数就够了。

| 名字 | 时间 | 它的主意 |
|-|-|-|
| Zep | 2025 年 1 月 | 把事实连成一张图，每条事实带着“从什么时候起有效、到什么时候失效”；新的事实进来，可以让旧的失效 |
| A-MEM | 2025 年 2 月 | 学卡片盒笔记法：每条记忆是一张卡片，新卡片进来，要和旧卡片连上线 |
| Mem0 | 2025 年 4 月 | 每来一条新信息，先和已有的比一比，再从四个动作里选一个：添加、更新、删除、不动 |
| MemOS | 2025 年 7 月 | 把记忆当成系统要统一管理的资源，每一份记忆都带着来源和版本 |

四家的做法差得很远。可你看它们多出来的那些东西：有效期，来源，版本，删除。全是冲着“现在有效”去的。

具体到写代码，更新一条记忆至少有三种完全不同的语义，不能混成同一个“覆盖”：

1. **纠正**：新事实推翻了旧结论（比如周一说“暂时没有实习生”，周三改成“有一位实习生”）。这时候不能把两条并排留着让人猜，也不能把旧的彻底抹掉——应当让新的替代当前状态，同时把旧的标成“曾经如此，后来更正”，留给事后审计。
2. **细化**：新信息把一个粗略的数拆细了（比如原来记着“27 张素材已复制”，后来核出“其中 25 张校验正常、2 张打不开”）。如果直接用后者整条覆盖前者，原来的总数 `27` 就丢了；细化应该补充粒度，而不是抹掉前一条仍然成立的事实。
3. **撤回**：用户明确要求删掉某条信息（比如周四让你忘掉手机尾号 `4471`）。只清掉 `device-note.md` 这一个文件远远不够：所有受控的副本、摘要、检索索引和交接包里都得同步清掉，而且不能让“查历史”再把一个已撤回的数翻回当前回答里。当然也要说实话：已经进到第三方日志或模型训练权重里的东西，靠应用层删不掉。

记清楚这三种更新方式之后，换一个问法。来接手的那一位，最先需要知道的到底是什么？

是这一周发生过什么吗？

## 一张只写“现在”的交接包

想想你自己请假之前是怎么交接的。

你不会把一周的聊天记录转给同事。你会写一张纸：事情办到哪儿了；哪些是弄好了但没人验过的；卡在哪儿，在等谁；他来了先干什么。就这四样。至于周一说过什么、周三怎么改的口，同事用不着知道。真要用了，他再来问你。

我们给资料助手也做这样一张纸，叫它**交接包**。四栏：

**现状。** 现在仍然有效的事实和决定。事情变了，就把旧的那条换掉，不让两条并排留着。

**待验证。** 做了，但还没有人确认做对了的事。

**阻塞。** 卡在哪儿，在等谁。

**下一步。** 接手的最先该做的事。

每一条后面都带着两样东西：哪一天记的，凭的是哪一句原话。

![[零基础学AI/attachments/第 11 章｜明天再见，它还接得上吗？/UyLibjB1goFFghxRFIWcePa8n4e.png|交接包有四栏。每一条都带着日期和一句原话。]]

“待验证”为什么要单占一栏？请回想第 08 章。那时候它说“已保存”，文件却并不存在，我们从此只认工具的回执。这里是同一件事再往后走一步：文件存进去了，回执也有了，这只说明**改了**。改得对不对是另一回事，要等有人照着走过一遍才知道。“改了”和“验证过了”中间隔着的那一段，在交接的时候最容易被抹平。

“阻塞”也是一样的道理。“在等组长点头”和“组长点头了”，只差一个人的一句话。这句话没到，就是没到。**等着，不会自己变成同意。**

谁来写这张纸？每天收工，让模型读一遍今天的对话，对着昨天的交接包提修改。

读到这儿你应该已经警觉了。刚才那件没人说过的待办，就是模型“顺手”写进摘要里的。让它来维护交接包，凭什么这一回就信它？

## 它提修改，程序对原话

不信它。照第 06 章的老规矩办：模型写，程序查。

模型不许直接改交接包。它只能递上来一条一条的修改：往哪一栏添什么，把哪一条去掉。每一条修改都必须附上**证据**，也就是从今天的对话里原样抄下来的一句话，一个字不许改。

程序收到修改，先去今天的对话里找这句话。找不到，退回，不写。

另外加一条。凡是“已验证”“走通了”“已同意”这类分量重的话，证据必须是**用户说的**。助手自己说的不算数。

被换下来的旧条目也不扔，另外存进一份流水。那是“当时如此”，留着备查。

```python
# handoff.py：第四种带法。不带“发生过什么”，只带“现在是什么样”。每天收工，模型提修改，程序核对过了才写进去
import json, os
from llm import chat, SPENT
from notebook import week, text

SECTIONS = ["现状", "待验证", "阻塞", "下一步"]

ASK = """你在替一件做了好几天的事维护一份“交接包”。接手的人不看以前的对话，只看这份交接包。
交接包有四栏：
- 现状：现在仍然有效的事实和决定。一件事一个 key；事情变了，就用同一个 key 覆盖掉旧的，不要另起一条。
- 待验证：做了、但还没有人确认做对了的事。文件存进去了，只说明“改了”；有人明说验证通过，才能把它从这一栏去掉。
- 阻塞：现在卡在哪儿，在等谁、等什么。“在等”不等于“已经同意”。
- 下一步：接手的人最先该做的事。
规矩：
1. 每一条都要有 quote：从今天的对话里原样抄一小段作证据，一个字都不许改。
2. 抄自公司资料的规定，要带上资料编号。
3. 用户说了不要保留的东西，不许写进来；已经在里面的，用 drop 去掉。
4. 已经不成立的条目，用 drop 去掉。没有变化的条目不用动。
下面是昨天收工时的交接包，和今天的全部对话。只输出 JSON：
{"patch": [{"op": "set", "section": "现状", "key": "短名字", "text": "一句话", "quote": "原话"},
           {"op": "drop", "section": "阻塞", "key": "短名字", "quote": "原话"}]}"""

SURE = ("已验证", "验证通过", "走通了", "已走通", "已批准", "已同意")     # 这类话，分量重

def squeeze(s):
    return "".join(str(s).split())

def check(op, pack, today, said):
    """一条修改能不能收。返回退回的理由；空的就是收下"""
    section, key = op.get("section"), op.get("key")
    if section not in SECTIONS or not key:
        return "栏目或名字不对"
    if op.get("op") == "drop" and key not in pack[section]:
        return "要去掉的这一条，包里本来就没有"
    quote = squeeze(op.get("quote", ""))
    if len(quote) < 6 or quote not in squeeze(today):
        return "证据在今天的对话里找不到原话"
    if op.get("op") == "set" and any(word in op.get("text", "") for word in SURE) and quote not in squeeze(said):
        return "“验证过”“同意了”这类话，只能拿用户的原话作证"
    return ""

def close(pack, past, date, messages):
    """收工。pack 是交接包，past 是被换下来的旧条目。返回 (收下几条, [(被退回的修改, 理由)])"""
    today = text(messages)
    said = "\n".join(m["content"] for m in messages if m["role"] == "user")
    reply = chat(ASK + "\n\n昨天收工时的交接包：\n" + render(pack, keys=True) + "\n\n今天的对话：\n" + today, json_mode=True)
    taken, refused = 0, []
    for op in json.loads(reply).get("patch", []):
        why = check(op, pack, today, said)
        if why:
            refused.append((op, why))
            continue
        section, key = op["section"], op["key"]
        if key in pack[section]:                                  # 被换下来的不扔，记到流水里：当时如此
            past.append({"section": section, "key": key, **pack[section].pop(key), "until": date})
        if op["op"] == "set":
            pack[section][key] = {"text": op["text"], "date": date, "quote": op["quote"]}
        taken += 1
    return taken, refused

def render(pack, keys=False):
    lines = []
    for section in SECTIONS:
        lines.append(f"【{section}】")
        for key, item in pack[section].items():
            lines.append(f"- {key + '：' if keys else ''}{item['text']}（{item['date']}，凭：“{item['quote']}”）")
        if not pack[section]:
            lines.append("- （空）")
    return "\n".join(lines)

def build(show=False):
    """把这一周从头到尾收五次工，返回 (交接包, 换下来的旧条目)"""
    pack, past = {section: {} for section in SECTIONS}, []
    for date, messages in week().items():
        taken, refused = close(pack, past, date, messages)
        if show:
            print(f"{date}收工：收下 {taken} 条修改，退回 {len(refused)} 条；包里现在 {sum(len(v) for v in pack.values())} 条")
            for op, why in refused:
                print(f"   退回〔{op.get('section')}〕{op.get('text', op.get('key'))}\n      理由：{why}。它给的证据：“{op.get('quote')}”")
    return pack, past

def now(question=None):
```

请把这两句对着读一遍。“卡在实习生那部分”，你说过。“正式员工部分通过”，你说过吗？

没有。这是模型从你的话里推出来的。推得也许没错：既然卡在后头，前头多半是过了的。可它终究是一个推断，却被写成了一条事实，旁边还端端正正附着一句真的原话。

> **作者手记：State-first 是怎么来的。** 这本书的作者平时同时用着好几个助手干活。它们来自不同的厂商，谁也读不到谁的对话。他给它们定的规矩里有这样几条：一个项目只有一份共用的记忆，哪个助手都不许另立一本私账；接手的时候先确认是哪个项目，再读这个项目的当前状态，然后才开工；状态里记的是目标、事实和证据、问题、在等什么、下一步；更早的经过全都留着，要用再翻，不一股脑塞进前文；收工时留下验证结果、待办和下一步。这一节的交接包，就是照着这套规矩缩小了做的。这种“先交代现在，再按需查过去”的次序，作者叫它 **State-first**。

```python
import json, os

def read_file(name):
    root = os.path.realpath("workspace")
    path = os.path.realpath(os.path.join(root, name))
    if os.path.dirname(path) != root or not os.path.isfile(path):
        return f"出错：workspace 里没有 {name}"
    return open(path, encoding="utf-8").read()

TOOLS = [tool("look_back", "查这件事以前发生过什么。交接包里没写、又需要知道的旧事，用它查。查到的是“当时如此”，不一定现在还有效。",

if __name__ == "__main__":
```

```python
"""

def work(date, lines):
    pack = json.load(open("memory/handoff.json", encoding="utf-8"))
    past = json.load(open("memory/handoff_past.json", encoding="utf-8"))
    records = json.load(open("memory/stream.json", encoding="utf-8"))
```

周五，这是模型的一个推断，带着一句真证据进了交接包。周一，新来的助手读到它，没有任何理由怀疑，把它当成事实，建议你对组长“直接说这部分已验证过”，又把“走通”两个字写进了要发给新同事的文件。

一个推断，每过一手就硬一分。

（收工的时候倒有个小插曲。新助手提的修改里有一条，开头是“老张已走通正式员工部分”，被程序退了回去，理由是“验证过”“同意了”这类话只能拿用户的原话作证。同一个意思，周五写成“通过”，溜过去了；周一写成“已走通”，撞上了。靠几个关键词把门，就是这个水平。）

## 趁没人用它的时候，对一遍原话

这样的故事，Anthropic 的研究者这个月刚讲过一个，主角是一个打游戏的 Claude。

2026 年 11 月 6 日，他们发表了一篇讲 Agent 记忆的文章：[The components of agent memory](https://www.anthropic.com/research/components-of-agent-memory)，里面有个实验。让模型玩《宝可梦》，一边玩一边给自己记笔记，笔记跨着一次次对话留下来。走到月见山的时候，它记错了一件事，把宝可梦中心记在了山的东边。凭着这条笔记，它走出山洞，认定自己已经翻过了山，在笔记里写下一句 Mt Moon DONE，然后从一道只能往下跳的坎上跳了下去，落回了来时的路。

带着这条笔记，它前后掉回去三次，最后也没能走到下一座城市华蓝市。

研究者接着做了一件事。他们让模型在不打游戏的时候，把存着那条错笔记的文件夹拿出来，对着当时的对话记录核了一遍。模型发现笔记和记录对不上，把它改了。带着改过的笔记重新出发，这一回它直接回到山里，最后走到了华蓝市。

这种趁闲着回头整理记忆的做法，文章里叫 dreaming，做梦。更早一些，2025 年 4 月，做 MemGPT 的那一组人（那时他们已经办了一家叫 Letta 的公司）发表过相近的想法，叫 sleep-time compute：让模型在没人提问的空当里，先把手头的材料想一遍。

同一篇文章里还有一组数，值得抄下来。两代模型各玩了一局。老的那一代留下 413 个笔记文件，拿到一枚徽章。新的那一代只留了 10 个文件，拿到四枚。各只有一局，当故事听就好。可它提醒了一件事：笔记多，不等于记得好。

我们的月见山，就是“正式员工部分通过”那半句话。照样做一遍。

```python
# python3 tidy.py 2 5    你看过以后，只照第 2、第 5 处疑点改
import json, sys
from llm import chat, SPENT
from notebook import week, text
from handoff import SECTIONS

ASK = """下面是交接包里的一条记录，和它所凭的那一天的全部对话。请对着对话核一遍：
这一条里，有没有哪个意思，是对话里没有人说过、工具也没有回过的？推断出来的、顺手补上的，都算。
只输出 JSON：{"ok": true 或 false, "extra": "多出来的那个意思；没有就留空", "fixed": "去掉它以后这一条该怎么写；ok 为 true 就留空"}"""

def doubt(pack):
    days, found = week(), []
    for section in SECTIONS:
        for key, item in pack[section].items():
            reply = json.loads(chat(ASK + "\n\n这一条：" + item["text"] + "\n\n那一天的对话：\n" + text(days[item["date"]]),
                                    json_mode=True))
            if not reply.get("ok") and reply.get("fixed"):
                found.append({"section": section, "key": key, "extra": reply["extra"], "fixed": reply["fixed"]})
    return found

if __name__ == "__main__":
    pack = json.load(open("memory/handoff.json", encoding="utf-8"))
    if not sys.argv[1:]:
        before = SPENT["tokens"]
        found = doubt(pack)
        json.dump(found, open("memory/doubts.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
        for n, d in enumerate(found, 1):
            print(f"疑点 {n}〔{d['section']}〕{pack[d['section']][d['key']]['text']}\n   多出来的：{d['extra']}\n   想改成：{d['fixed']}\n")
        print(f"核了 {sum(len(pack[s]) for s in SECTIONS)} 条，有疑点的 {len(found)} 条，花了 {SPENT['tokens'] - before} 个 token。还什么都没改。")
    else:
        found = json.load(open("memory/doubts.json", encoding="utf-8"))
        for n in map(int, sys.argv[1:]):
            d = found[n - 1]
            pack[d["section"]][d["key"]]["text"] = d["fixed"]
            print(f"照疑点 {n} 改了：{d['fixed']}")
        json.dump(pack, open("memory/handoff.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
```

从第 03 章数过来，它“知道”的事一共住在五个地方。

模型里学到的。训练时定下的，你改不了，它也不知道你们组的事。

这一次的前文。摆在它眼前的全部文字，对话一关就没了。

本子。每天的对话原样存着。这一章开头我们把它整本读回去，后来只留着备查。

旧记录。从本子里一句一条记下来的事，按需去查。

交接包。这件事现在是什么样，每次开工先读。

越往下，离“这一件事”越近，变得越快，也越要有人盯着改。

还有两样东西，这一章一直小心地没往记忆里放。

一样是公司的规定。实习生审批是 2 个工作日，这句话的家在资料库里（第 07 章）。记忆里可以记“清单是按 D03 写的”，带着编号，日后好重查。规定本身不该抄一份过来，抄过来的那一份，规定一改就成了错的。

另一样是做法。“更新检查分几步”“改文件前先读原文”住在 Skill 卡片上（第 10 章）。

这样一来，范围就分清了：**个人长期记忆**（跨任务依然有效的偏好与约束，例如你的工号或汇报风格）、**当前任务状态**（留在交接包里，事情办完就归档，比如“day1.md 还没验过”），和 **Skill 做法**（留在卡片上，不跟某一次任务绑定），三样东西各有各的家。上一章末尾那张图上用虚线画的第三格，现在可以描实了。

开头那一注，也可以全揭开了。把所有对话都存下来，算不算有了记忆？

存下来，只是有了本子。**记忆是另外三件事：记什么，什么时候拿出来，什么时候改掉。** 小镇认真回答了第二件。交接包想回答第三件，答得还不算好。

**选读：这一章的两个名字。** “读不下了，就压成摘要”那一步，业内叫 **compaction**（压缩）。能长时间干活的 Agent 都有这一手：前文快满的时候，自动把较早的部分压成一份摘要，腾出地方接着干。你在这一章见过它的代价：没人说过的待办会冒出来，说过的细节会丢。所以讲究的做法里，压缩之前要先把要紧的东西写进交接包。更大的那个名字叫**上下文工程**：琢磨每一次开工该让它看见什么、不该让它看见什么。这一章从头到尾在做的就是这件事。前两年人人在谈“提示词怎么写”，这两年谈得更多的是“上下文怎么装”。

## 这一章你留下了什么

`days.py`、`week.py` 和 `log` 文件夹里的本子，是一周的原始记录。以后你想试别的记法，可以直接拿它们来考。

`handover.py` 是接手考卷。和第 01 章的 `exam.py`、第 07 章的 `quiz.py` 一样：先出题，后写程序。

`full.py`、`summary.py`、`stream.py`、`handoff.py` 是四种带法，`arena.py` 让它们同场。

`memory_server.py`、`work.py`、`tidy.py`，加上原来的 `assistant.py`，是资料助手的 **v1.1**：开工先读交接包；旧事按需去查；收工时模型提修改，程序对原话；闲下来把交接包对着本子核一遍，改哪条由你定。

`forge.py` 和 `sunk.py` 是两个不花钱的小实验。一个告诉你那道检查拦得住什么，一个告诉你记忆流会把什么沉下去。

v1.1 带着三处没治好的毛病，都摆在明面上了。一条里装两件事，覆盖的时候会丢。“下一步”那一栏里有过期的条目，没人清。推断能披着真证据混进来。

还有五句话。

存下来的是本子，还不是记忆。

当时如此，不等于现在有效。

改了，不等于验证了。等着，不等于同意了。

证据只能证明那句话有人说过。

笔记少而准，胜过多而杂。

```python
# notebook.py：翻本子用的两个小工具。按天拆开；把一段对话印成人能读的文字
import json
from days import WEEK, load

def by_day(talk):
    """整本对话，按每句话开头的日期拆成几天：{日期: [那一天的消息]}"""
    days, today = {}, None
    for message in talk[1:]:                              # 第一条是它的角色说明，不算哪一天的
        if message["role"] == "user" and message["content"].startswith("（"):
            today = message["content"][1:9]
        days.setdefault(today, []).append(message)
    return days

def text(messages):
    """把消息印成文字：谁说了什么，用了什么工具，工具回了什么"""
    lines = []
    for m in messages:
        if m["role"] == "user":
            lines.append("用户：" + m["content"])
        elif m["role"] == "tool":
            lines.append("工具回：" + m["content"])
        else:
            for call in m.get("tool_calls") or []:
                lines.append(f"助手用工具：{call['function']['name']} {call['function']['arguments']}")
            if m.get("content"):
                lines.append("助手：" + m["content"])
    return "\n".join(lines)

def week():
    return by_day(load(list(WEEK)[-1]))                   # 周五收工时的本子，就是这一整周

if __name__ == "__main__":
    for date, messages in week().items():
        print(date, len(messages), "条，", len(text(messages)), "个字")
```

```python
# arena.py：擂台。同一周，同一张考卷，四种带法各从头整理三遍、各考三遍；最后再加一位：交接包 + 按需查旧事
import contextlib, io, json, os
from llm import SPENT
from handover import grade
import full, summary, stream, handoff

def quiet(label, material_for):
    with contextlib.redirect_stdout(io.StringIO()):
        return grade(label, material_for)

ROUNDS, table, kept = 3, {}, []
for n in range(ROUNDS):
    before = SPENT["tokens"]
    table.setdefault("全文", []).append((*quiet("全文", full.whole_week), 0))

    before = SPENT["tokens"]
    notes = summary.build()
    cost = SPENT["tokens"] - before
    table.setdefault("摘要", []).append((*quiet("摘要", lambda q: notes[list(notes)[-1]]), cost))

    before = SPENT["tokens"]
    records = stream.build()
    cost_stream = SPENT["tokens"] - before
    table.setdefault("检索", []).append((*quiet("检索", lambda q: stream.recall(q, records)), cost_stream))

    before = SPENT["tokens"]
    pack, past = handoff.build()
    cost_pack = SPENT["tokens"] - before
    table.setdefault("交接包", []).append((*quiet("交接包", lambda q: handoff.render(pack)), cost_pack))

    both = lambda q: handoff.render(pack) + "\n\n【按这个问题，从旧记录里查到的三条】\n" + stream.recall(q, records, k=3)
    table.setdefault("交接包＋按需查", []).append((*quiet("交接包＋按需查", both), cost_pack + cost_stream))
    kept.append({"summary": notes, "stream": records, "pack": pack, "past": past})
    print(f"第 {n + 1} 轮考完")

os.makedirs("memory", exist_ok=True)
json.dump(kept, open("memory/arena_rounds.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
print(f"\n{'带法':<10}{'三轮各答对几道':<14}{'每道题要读':<12}{'一周收工整理花掉'}")
for label, rows in table.items():
    scores = "  ".join(str(r[0]) for r in rows)
    read = sum(r[1] for r in rows) // ROUNDS
    spent = sum(r[2] for r in rows) // ROUNDS
    print(f"{label:<10}{scores + ' / 10':<18}{read:>6} token   {spent:>8} token")
```

```python
# assistant.py：资料助手。工具从服务那儿接，做法从卡片上取，和上一章一样
# 这一章多了两个口子：talk，把一个本子交给它接着写；extra，把一段话额外摆进它的前文
import json
from host import connect, ask_human, ROLE
from skills import shelf, catalog, LOAD_SKILL
from agent import agent

SERVERS = ["docs_server.py", "archive_server.py"]

def build(card_roots=("skills",), confirm=ask_human, extra="", servers=SERVERS):
    """返回两样：ask(一句话, talk) 让它办事；plugs 是接着的服务"""
    tools, run, plugs = connect(servers, confirm)
    cards = shelf(*card_roots)

    def run_all(call):
        if call["function"]["name"] != "load_skill":
            return run(call)
        name = json.loads(call["function"]["arguments"]).get("name")
        return cards[name][1] if name in cards else f"出错：没有 {name} 这张卡片"

    def ask(goal, talk=None, **more):
        return agent(goal, tools + [LOAD_SKILL], run_all, ROLE + catalog(cards) + extra, talk=talk, **more)

    return ask, plugs
```

`agent.py` 只改了两行：交给它一个空本子的时候，它会先把自己的角色写在第一页。别的和上一章一样，不再贴了。
